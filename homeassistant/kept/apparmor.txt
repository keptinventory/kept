#include <tunables/global>

# Kept's Home Assistant app (D272). In complain mode until the Home Assistant OS run reads its
# audit log (`journalctl _TRANSPORT="audit" -g 'apparmor="ALLOWED"'`) after using every feature,
# then enforced (the security guide's method). The rules marked "complain run" cover what the
# 2026-10-29 run on aarch64 logged (setup, a backup, an uninstall and restore, a reboot). The
# launch (docs/runbooks/release.md, W3b) drops `complain` only after a run in enforce mode shows
# zero denials.
profile kept flags=(attach_disconnected,mediate_deleted,complain) {
  #include <abstractions/base>
  #include <abstractions/nameservice>
  #include <abstractions/openssl>

  # The start script re-owns /data and drops to postgres and to Kept's user (setpriv).
  capability chown,
  capability dac_override,
  capability dac_read_search,
  capability fowner,
  capability setgid,
  capability setuid,
  capability kill,
  # Complain run: install, chmod and sed in the start script asked for fsetid (keeping a set-group-ID
  # bit on a file whose group isn't theirs). Refused, the kernel only clears that bit, which nothing
  # here uses: so it is denied, quietly, rather than allowed.
  deny capability fsetid,

  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network unix,

  signal (send,receive) peer=@{profile_name},
  signal (receive) peer=unconfined,

  # Docker's init (config.yaml `init: true`), the start and backup scripts, the shell tools they use.
  /sbin/docker-init ix,
  /dev/init ix,
  /usr/local/bin/kept-ha-start rix,
  /usr/local/bin/kept-ha-backup rix,
  /usr/local/bin/kept rix,
  /usr/local/bin/node ix,
  /usr/local/bin/restic ix,
  /usr/bin/ssh ix,
  /{usr/,}bin/** ix,
  /{usr/,}sbin/** ix,
  /usr/lib/postgresql/18/bin/* ix,
  /usr/lib/postgresql/18/lib/** mr,
  # Complain run: sharp's native module and its libvips, mapped by Kept (photo resizing). arm64
  # was logged; x64 is the same package for amd64.
  # Quoted, so no "@" is read as the start of a variable (@{...}).
  "/app/node_modules/.pnpm/@img+sharp-linux-{arm64,x64}@*/node_modules/@img/sharp-linux-{arm64,x64}/lib/*.node" mr,
  "/app/node_modules/.pnpm/@img+sharp-libvips-linux-{arm64,x64}@*/node_modules/@img/sharp-libvips-linux-{arm64,x64}/lib/lib*.so*" mr,

  # Read-only: the app, the system, Home Assistant's certificates.
  /app/** r,
  # Complain run: find (the start script's) opens its working directory, /app.
  /app/ r,
  /usr/** r,
  /etc/** r,
  /ssl/** r,
  /usr/local/lib/kept-ha/** r,
  /var/lib/postgresql/** r,
  /proc/** r,
  /sys/fs/cgroup/** r,
  /sys/devices/system/** r,

  # Everything that lasts, and what doesn't.
  # Complain run: the start script chowns and chmods /data itself (AppArmor's `w` on it); it
  # creates nothing there through this rule, since a new file is checked by its own path.
  /data/ rw,
  /data/** rwk,
  # Complain run: atomic writes (a temporary file, then a link to the final name) for Kept's keys,
  # the database logins' passwords and the owner login's.
  /data/config/** l,
  /data/owner/* l,
  /tmp/** rwlk,
  /run/** rwk,
  /dev/shm/** rwk,
  # Complain run, 11 times, logged as:
  #   apparmor="ALLOWED" operation="file_mmap" class="file" profile="local_kept" name="/"
  #   comm="postgres" requested_mask="wr" denied_mask="wr"
  # A shared mapping with no path of its own, which attach_disconnected names "/". Likely
  # Postgres's anonymous MAP_HUGETLB attempt (huge_pages=try, the default), now off in
  # kept-ha-start (`-c huge_pages=off`, for initdb too). This rule stays until the next enforced
  # run shows whether the mapping still happens: it is that one exact path, not a glob, and grants
  # nothing under /. If that run logs no "/", it goes.
  / rw,
  /dev/null rw,
  /dev/zero r,
  /dev/urandom r,
  /dev/tty rw,
}
